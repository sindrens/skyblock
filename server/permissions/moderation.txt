# LuckPerms grants for the moderation module. Applied by scripts/apply-permissions.sh after base.txt (groups exist by then).
# Node definitions: modules/moderation/src/main/resources/permissions/moderation.yml. One console command per line; # = comment.

# Moderator: unban.
lp group mod permission set skyblock.staff.unban true

# --- Punishments (M1.8). Length limits: moderation.yml limits.max-duration (helper 1d, mod 30d, admin perm) ---
lp group helper permission set skyblock.staff.punish.warn true
lp group helper permission set skyblock.staff.punish.kick true
lp group helper permission set skyblock.staff.punish.mute true
lp group helper permission set skyblock.staff.punish.gui true
lp group helper permission set skyblock.staff.punish.history true
lp group helper permission set skyblock.staff.punish.max.helper true
lp group mod permission set skyblock.staff.punish.ban true
lp group mod permission set skyblock.staff.punish.unmute true
lp group mod permission set skyblock.staff.punish.unwarn true
lp group mod permission set skyblock.staff.punish.alts true
lp group mod permission set skyblock.staff.punish.silent true
lp group mod permission set skyblock.staff.punish.max.mod true
lp group admin permission set skyblock.staff.punish.ipban true
lp group admin permission set skyblock.staff.punish.max.admin true
lp group admin permission set skyblock.staff.punish.exempt true
lp group admin permission set skyblock.staff.punish.exempt.bypass true

# --- Phase P, P2: undo, double-punish protection, /pstats ---
# Every staff member sees their own numbers; admins see everyone's and the ranking, may use -force and undo others.
lp group helper permission set skyblock.staff.punish.stats true
lp group admin permission set skyblock.staff.punish.stats.others true
lp group admin permission set skyblock.staff.punish.force true
lp group admin permission set skyblock.staff.punish.undo.others true

# --- Phase P, P3: staff notes. /mypunishments and /ack (skyblock.punish.view.own) are default: true for everyone ---
lp group helper permission set skyblock.staff.punish.notes true

# --- Phase P, P4: approval for big punishments, watchlist ---
# Admins approve requests (ban perm, ipban, mute perm, ban >= 30d, or longer than the rank limit); helpers and mods
# create requests. Every staff member can watch players; the join line goes to skyblock.staff.notify.
lp group admin permission set skyblock.staff.punish.approve true
lp group helper permission set skyblock.staff.watch true

# --- Phase P, P5: bulk punish, appeals. /appeal (skyblock.punish.appeal) is default: true for everyone ---
# Bulk is senior staff (mods and up). Appeals are handled by mods and up (accepting lifts a mute or ban, which helpers
# can't do with /unmute or /unban either). New appeals are announced to skyblock.staff.notify; old open ones
# (extras.appeal.reminder) are listed on join to skyblock.staff.appeals.
lp group mod permission set skyblock.staff.punish.bulk true
lp group mod permission set skyblock.staff.appeals true

# --- Phase P, P6: raid lockdown, anticheat suggestions ---
# Every rank above default is never affected by a chat lockdown (spec 13.9); mod/admin/owner inherit it from helper.
lp group ember permission set skyblock.lockdown.bypass true
lp group builder permission set skyblock.lockdown.bypass true
lp group helper permission set skyblock.lockdown.bypass true
# Every staff member can start/end a lockdown (it only limits new accounts and ends by itself; open question for the owner).
lp group helper permission set skyblock.staff.lockdown true
# /psuggest for staff; GrimAC runs it from the console (which has every permission).
lp group helper permission set skyblock.staff.punish.suggest true
