# LuckPerms grants for the profiles module. Applied by scripts/apply-permissions.sh after base.txt (groups exist by then).
# Node definitions: modules/profiles/src/main/resources/permissions/profiles.yml. One console command per line; # = comment.

# /profiles is for everyone (plugin default: true).
# Free profile slots come with Firewing (slot 2) and Pyrelord (slot 3): granted in base.txt (Ranks V2 section) (card F2, RANKS-V2).

# Staff (feedback 31 Oct 2026): helpers see a player's profiles, admins can reset one (offline, confirm, audited).
lp group helper permission set skyblock.profiles.admin.info true
lp group admin permission set skyblock.profiles.admin.reset true

# 4 Nov 2026 (island = profile): leaving an island wipes the profile into a 14-day snapshot. Moderators see the snapshots,
# admins put one back (mistakes, griefing kicks).
lp group mod permission set skyblock.profiles.admin.snapshots true
lp group admin permission set skyblock.profiles.admin.restore true
