# LuckPerms grants for the economy module. Applied by scripts/apply-permissions.sh after base.txt (groups exist by then).
# Node definitions: modules/economy/src/main/resources/permissions/economy.yml. One console command per line; # = comment.
# Player nodes (/bal, /baltop, /pay, /transactions, /withdraw, /sell, /bank) default to true, so they need no grants.
# Island bank rights are island role permissions (/is permissions: bank-deposit, bank-withdraw, bank-history), not nodes.
# Dupe alerts for copied money notes use skyblock.staff.notify (granted in base.txt with the other staff alerts).

# --- Economy (staff) ---
lp group helper permission set skyblock.staff.economy.alerts true
# /eco give|take|set|reset|note: admins and the owner (owner decision #47; owner inherits admin). Always audited and
# staff-alerted. The old owner node stays granted so nothing changes for the owner if inheritance is ever removed.
lp group admin permission set skyblock.admin.economy true
lp group owner permission set skyblock.owner.economy true
# M3.8 economy dashboard (/eco admin) and anomaly alerts: moderators and up (admin and owner inherit mod; admins also get
# the dashboard as a child of skyblock.admin.economy). Read-only plus alert review; alerts never punish anyone.
lp group mod permission set skyblock.staff.economy.dashboard true
lp group mod permission set skyblock.staff.economy.anomalies true

# Owner answer 334: staff accounts (builders, helpers and up) play normally; their big transfers alert admins.
lp group builder permission set skyblock.staff.economy.flagged true
lp group helper permission set skyblock.staff.economy.flagged true
