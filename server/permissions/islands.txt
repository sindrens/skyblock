# LuckPerms grants for the islands module. Applied by scripts/apply-permissions.sh after base.txt (groups exist by then).
# Node definitions: modules/islands/src/main/resources/permissions/islands.yml. One console command per line; # = comment.

# Islands (M2.1). Everyone can use /is (plugin default true).
# Members and roles (M2.3: invite, accept, deny, kick, leave, promote, demote, transfer, coop, uncoop, members) use the
# same node; who may do what is decided by the island role, not by LuckPerms.
# Staff protection bypass (M2.2): /is bypass toggles it, every toggle is logged (audit category staff).
# Admins only for now (open question: also moderators?).
lp group admin permission set skyblock.island.bypass true

# Visiting (M2.4): /is visit, lock, unlock, expel, ban, unban, setwarp, delwarp, warp, rate, like use skyblock.island
# (everyone); the island role decides who may lock, ban and manage warps. Staff enter locked islands and islands that
# banned them (needed for /isadmin tp and reports); helper < mod < admin, so everyone above inherits it.
lp group helper permission set skyblock.island.visit.bypass true

# Island staff tools (M2.7, /isadmin; every use is logged in category staff). Helpers look (info, tp); admins change.
# helper < mod < admin, so mods and admins inherit the helper nodes.
lp group helper permission set skyblock.island.admin true
lp group helper permission set skyblock.island.admin.info true
lp group helper permission set skyblock.island.admin.tp true
lp group admin permission set skyblock.island.admin.delete true
lp group admin permission set skyblock.island.admin.setsize true
lp group admin permission set skyblock.island.admin.addmember true
lp group admin permission set skyblock.island.admin.kick true
lp group admin permission set skyblock.island.admin.setowner true
lp group admin permission set skyblock.island.admin.reset true
lp group admin permission set skyblock.island.admin.slots true
lp group admin permission set skyblock.island.admin.teleport-slot true

# Islands round 2 (ISL2): /is settings, /is setmyhome, /is invites use skyblock.island (everyone); the island role and the
# "settings" role permission decide who may change settings. Island fly (skyblock.island.fly) is a perk: no rank gets it
# yet (open question: which ranks?); staff with * have it. The Wings mastery will give fly per island without this node.

# Island notes (NOTES): /is notes uses skyblock.island (everyone); the island role permissions notes-read, notes-write and
# notes-manage decide who reads and writes. Staff read any island's notes and recently deleted ones (owner N7: for scam
# and abuse reports); every staff look is logged (category staff). helper < mod < admin.
lp group helper permission set skyblock.island.notes.staff true

# Island menu and island chat (M2.5): /is (menu), /is chat and /is info use skyblock.island (everyone); island chat is for
# stored members (not coop/visitors). Staff spy on island chat with the "Island chat spy" /settings toggle (default off),
# like /socialspy. helper < mod < admin.
lp group helper permission set skyblock.island.chat.spy true

# Island stats (M2.6): /is stats uses skyblock.island.stats (plugin default true, everyone). Staff look at any island's
# stats (helpers and up); only admins reset them (confirm; logged in category staff as isadmin_stats_reset).
lp group helper permission set skyblock.island.admin.stats true
lp group admin permission set skyblock.island.admin.stats.reset true
