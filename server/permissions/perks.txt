lp group mod permission set skyblock.staff.repair true
lp group mod permission set skyblock.staff.repair.others true
lp group mod permission set skyblock.staff.nick.others true
lp group admin permission set skyblock.admin.sudo false
lp group owner permission set skyblock.admin.sudo true
