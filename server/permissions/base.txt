# Staff ranks for SkybornMC: groups, weights, colors, prefixes, inheritance, third-party plugins, wildcards, people.
# Applied first by scripts/apply-permissions.sh after every update/deploy, then every other *.txt here (one per
# module, alphabetical). One console command per line. Lines starting with # are ignored. Every command is safe to run repeatedly.
# Donor ranks and the three won ranks are in ranks.txt (applied after this file, RANKS-V2). Every rank above default has a
# weight above 0: lockdowns exempt every rank above default (owner answer 460).

# --- Groups and weights (higher = more important). Staff 600-999, donors 100-160 (ranks.txt), free 10, default 0. ---
lp creategroup builder
lp creategroup helper
lp creategroup mod
lp creategroup admin
lp creategroup owner
lp group default setweight 0
lp group builder setweight 600
lp group helper setweight 700
lp group mod setweight 800
lp group admin setweight 900
lp group owner setweight 999
# Free rank Ashborn (group id stays ember, owner EP-Q11/EP-Q15): given automatically after the tutorial and 30 minutes.
lp creategroup ember
lp group ember setweight 10
lp group ember parent add default
lp group ember setdisplayname Ashborn
lp group default setdisplayname New

# --- Rank colours (hex) = fallback only: SAME hex as chat.yml ranks.looks ---
lp group default meta set color "#555555"
lp group ember meta set color "#AAAAAA"
lp group builder meta set color "#FFAA00"
lp group helper meta set color "#55FF55"
lp group mod meta set color "#55FFFF"
lp group admin meta set color "#FF5555"
lp group owner meta set color "#FFFF55"

# --- Display names (the rank word) and prefixes (priority = weight; old priorities are removed first) ---
lp group builder setdisplayname BUILDER
lp group helper setdisplayname HELPER
lp group mod setdisplayname MOD
lp group admin setdisplayname ADMIN
lp group owner setdisplayname OWNER
lp group ember meta removeprefix 1
lp group ember meta removeprefix 10
lp group ember meta addprefix 10 "&7Ashborn &f"
lp group builder meta removeprefix 5
lp group builder meta removeprefix 600
lp group builder meta addprefix 600 "&6&lBUILDER &f"
lp group helper meta removeprefix 10
lp group helper meta removeprefix 700
lp group helper meta addprefix 700 "&a&lHELPER &f"
lp group mod meta removeprefix 20
lp group mod meta removeprefix 800
lp group mod meta addprefix 800 "&b&lMOD &f"
lp group admin meta removeprefix 30
lp group admin meta removeprefix 900
lp group admin meta addprefix 900 "&c&lADMIN &f"
lp group owner meta removeprefix 100
lp group owner meta removeprefix 999
lp group owner meta addprefix 999 "&e&lOWNER &f"

# --- Inheritance: helper < mod < admin < owner. Admin can also build. ---
lp group helper parent add default
lp group mod parent add helper
lp group admin parent add mod
lp group admin parent add builder
lp group owner parent add admin

# --- Builder: build spawn, the Depths and the warzone ---
lp group builder permission set worldedit.* true
lp group builder permission set fawe.* true
lp group builder permission set minecraft.command.gamemode true
lp group builder permission set minecraft.command.teleport true
lp group builder permission set minecraft.command.time true
lp group builder permission set skyblock.admin.worlds true

# --- Helper: see cheat alerts (staff chat: chat.txt) ---
lp group helper permission set grim.alerts true

# --- Moderator: moderation and teleporting ---
lp group mod permission set minecraft.command.kick true
lp group mod permission set minecraft.command.ban true
lp group mod permission set minecraft.command.pardon true
lp group helper permission set skyblock.staff.notify true
# VIS: staff stays visible to everyone while not vanished (players' Show players choice, the AFK pool).
lp group helper permission set skyblock.staff.visible true
lp group mod permission set minecraft.command.teleport true
lp group mod permission set minecraft.command.gamemode true
lp group mod permission set grim.brand true
lp group mod permission set coreprotect.inspect true
lp group mod permission set coreprotect.lookup true

# Plugin-level nodes (plugin.yml): config editor. Owner-level config comes from owner's *.
lp group admin permission set skyblock.admin.config true

# --- Admin: server administration (not permission editing) ---
lp group admin permission set skyblock.admin.* true
lp group admin permission set minecraft.command.* true
lp group admin permission set grim.* true
lp group admin permission set coreprotect.* true
lp group admin permission set viaversion.admin true
lp group admin permission set geyser.command.* true
lp group admin permission set floodgate.command.* true

# Every staff tool (incl. editing inventories, /give, /skull, /more).
lp group admin permission set skyblock.staff.* true

# --- Owner: everything ---
lp group owner permission set * true

# --- People (add rank holders here: lp user <name> parent add <rank>) ---
lp user Pouched parent add owner
