# Staff ranks for SkybornMC. Applied automatically by scripts/apply-permissions.sh after every update/deploy.
# One console command per line. Lines starting with # are ignored. Every command is safe to run repeatedly.
# Donor ranks (Squire -> Monarch) are added when the store/ranks phase is built.

# --- Groups and weights (higher = more important) ---
lp creategroup builder
lp creategroup helper
lp creategroup mod
lp creategroup admin
lp creategroup owner
lp group default setweight 0
lp group builder setweight 5
lp group helper setweight 10
lp group mod setweight 20
lp group admin setweight 30
lp group owner setweight 100

# --- Rank colors (hex) for our chat/tab/nametags; prefixes below stay classic colors for Bedrock and other plugins ---
lp group default meta set color "#C9C9C9"
lp group builder meta set color "#FFD23F"
lp group helper meta set color "#63B8FF"
lp group mod meta set color "#FF8A1F"
lp group admin meta set color "#FF3D2E"
lp group owner meta set color "#C21F2B"

# --- Display names and chat prefixes (Phoenix palette, Bedrock-safe classic colors) ---
lp group builder setdisplayname Builder
lp group helper setdisplayname Helper
lp group mod setdisplayname Moderator
lp group admin setdisplayname Admin
lp group owner setdisplayname Owner
lp group builder meta removeprefix 5
lp group builder meta addprefix 5 "&e&lBUILDER &e"
lp group helper meta removeprefix 10
lp group helper meta addprefix 10 "&b&lHELPER &b"
lp group mod meta removeprefix 20
lp group mod meta addprefix 20 "&6&lMOD &6"
lp group admin meta removeprefix 30
lp group admin meta addprefix 30 "&c&lADMIN &c"
lp group owner meta removeprefix 100
lp group owner meta addprefix 100 "&4&lOWNER &4"

# --- Inheritance: helper < mod < admin < owner. Admin can also build. ---
lp group helper parent add default
lp group mod parent add helper
lp group admin parent add mod
lp group admin parent add builder
lp group owner parent add admin

# --- Builder: build spawn, the Depths and the warzone ---

lp group builder permission set minecraft.command.teleport true
lp group builder permission set minecraft.command.time true

lp group builder permission set axiom.entity.* true
lp group builder permission set axiom.blueprint.* true

lp group mod permission set minecraft.command.gamemode true
